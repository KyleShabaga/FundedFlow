use serde_json::{from_value, json, Value};

use crate::csv_import;
use crate::db;
use crate::store::{accounts, journal, trades};

fn account(conn: &mut rusqlite::Connection, v: Value) -> String {
    accounts::create(conn, from_value(v).unwrap()).unwrap().id
}

fn trade(conn: &mut rusqlite::Connection, v: Value) -> String {
    trades::create(conn, from_value(v).unwrap()).unwrap().id
}

const CSV: &str = "symbol,_priceFormat,_priceFormatType,_tickSize,buyFillId,sellFillId,qty,buyPrice,sellPrice,pnl,boughtTimestamp,soldTimestamp,duration

fn fixture_gen() {
    let out = std::env::var("FIXTURE_OUT").expect("FIXTURE_OUT");
    let _ = std::fs::remove_file(&out);
    let mut conn = db::open_file(std::path::Path::new(&out)).unwrap();

    let base = |name: &str, tz: &str, reset: &str, r: &str| {
        json!({"name": name, "broker": "Tradovate", "propFirm": "Apex", "externalRef": r,
               "accountType": "evaluation", "status": "active", "currency": "USD",
               "startingBalance": "50000", "timezone": tz, "tradingDayResetTime": reset, "notes": null})
    };
    let ny = account(&mut conn, base("NY 17:00 eval", "America/New_York", "17:00", "APEX-1"));
    let utc = account(&mut conn, base("UTC calendar", "Etc/UTC", "00:00", "APEX-2"));

    let t = |acc: &str, sym: &str, side: &str, entry: &str, exit: Option<&str>, notes: &str| {
        json!({"accountId": acc, "symbol": sym, "side": side, "quantity": "1",
               "entryAt": entry, "entryPrice": "6000.25",
               "exitAt": exit, "exitPrice": exit.map(|_| "6004.5"),
               "fees": "1.24", "realizedPnl": null, "notes": notes})
    };
    // Mon 10:30 New York: Monday under every rule.
    let mon = trade(&mut conn, t(&ny, "MESZ6", "long", "2026-10-19T14:00:00Z", Some("2026-10-19T14:30:00Z"), "mon"));
    // Fri 17:30 New York (after the 17:00 reset): Saturday before the weekend roll, Monday after.
    trade(&mut conn, t(&ny, "MESZ6", "short", "2026-10-16T21:00:00Z", Some("2026-10-16T21:30:00Z"), "fri after reset"));
    // Sun 18:00 New York: next day = Monday under every rule.
    trade(&mut conn, t(&ny, "ESZ6", "long", "2026-10-18T22:00:00Z", Some("2026-10-18T22:10:00Z"), "sun evening"));
    // Open trade entered Sat 10:00 New York.
    trade(&mut conn, t(&ny, "MNQZ6", "long", "2026-10-17T14:00:00Z", None, "open sat"));
    // Soft-deleted Fri 18:00 New York.
    let del = trade(&mut conn, t(&ny, "MESZ6", "long", "2026-10-16T22:00:00Z", Some("2026-10-16T22:05:00Z"), "deleted"));
    trades::delete(&mut conn, &del).unwrap();
    // Reported-P&L fallback for an unknown symbol, Sat 10:00 UTC.
    let mut unknown = t(&utc, "XYZ", "long", "2026-10-17T09:00:00Z", Some("2026-10-17T10:00:00Z"), "unknown sat");
    unknown["realizedPnl"] = json!("12.5");
    trade(&mut conn, unknown);
    // Thursday, calendar day.
    trade(&mut conn, t(&utc, "MESZ6", "long", "2026-10-15T09:00:00Z", Some("2026-10-15T10:00:00Z"), "thu"));

    

    csv_import::commit(
        &mut conn,
        &from_value(json!({"accountId": ny, "fileName": "Performance.csv", "content": CSV,
                           "timezone": "America/New_York"}))
        .unwrap(),
    )
    .unwrap();

    journal::create(
        &mut conn,
        from_value(json!({"accountId": ny, "tradeId": mon, "entryAt": "2026-10-19T15:00:00Z",
                          "title": "Monday review", "body": "Waited for the open.", "tags": ["patience", "es"]}))
        .unwrap(),
    )
    .unwrap();

    let arch = account(&mut conn, base("Archived", "America/New_York", "17:00", "APEX-3"));
    trade(&mut conn, t(&arch, "MESZ6", "long", "2026-10-16T21:00:00Z", Some("2026-10-16T21:30:00Z"), "archived fri"));
    accounts::archive(&conn, &arch).unwrap();

    drop(conn);
}
